QUICKSORT(A, p, r)
if p < r
    q = PARTITION(A, p, r)
    QUICKSORT(A, p, q - 1)
    QUICKSORT(A, q + 1, r)


PARTITION(A, p, r)
x = A[r]
i = p - 1
for j = p to r - 1
    if A[j] <= x
        i = i + 1
        swap A[i] with A[j]
swap A[i + 1] with A[r]
return i + 1


RANDOMIZED_PARTITION(A,p,r)
if p < r
    q = RANDOMIZED_PARTITION(A, p, r)
    RANDOMIZED_QUICKSORT(A, p, q - 1)
    RANDOMIZED_QUICKSORT(A, q + 1, r)


RANDOMIZED_PARTITION(A, p, r)
i = RANDOM(p, r)
swap A[i] with A[r]
return PARTITION(A, p, r)
