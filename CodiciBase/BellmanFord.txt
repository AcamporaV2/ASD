BELLMAN-FORD(G,w,s)

for each u in V[G] - {s}
    d[u] = infinito
    padre[u] = NIL

d[s] = 0


for i = 1 to V[G]-1
    for each (u,v) in E[G]

        if d[v] > d[u] + w(u,v)
            d[v] = d[u] + w(u,v)
            padre[v] = u 

//ciclo negativo
for each (u,v) in E[G]
    if d[v] > d[u] + w(u,v)
        return FALSE 

Return TRUE
