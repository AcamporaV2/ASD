HASH-INSERTI(T,k)
i = 0

repeat 

    j = h(k,i)
    if T[j] = NIL 
        T[j] = k 
        return j 

    else 
        i = i+1

until i = m 


HASH-SEARCH(T,k)

i = 0
 
repeat 
    j = h(k,i)
    if T[j] = k 
        return j 
    
    i = i+1

until T[j] = NIL or i = m

HASH-DELETE(T,k)

i = 0

repeat 
    j = h(k,i)
    if T[j] = k 
        T[j] = DELETED
        return j
    
    i = i+1

until T[j] = NIL or i = m
