LOOKUP-LCS(X, Y, i, j, c)
  if c[i, j] != -1
      return c[i, j]
  if i == 0 or j == 0
      c[i, j] = 0
  else if X[i] == Y[j]
      c[i, j] = 1 + LOOKUP-LCS(X, Y, i-1, j-1, c)
  else
      c[i, j] = max(LOOKUP-LCS(X, Y, i-1, j, c), LOOKUP-LCS(X, Y, i, j-1, c))
  return c[i, j]
