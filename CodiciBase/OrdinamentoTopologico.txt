TOPOLOGICAL-SORT(G)
1  // Inizializzazione della lista che conterrà l'ordinamento
2  Linked_List = ∅
3  
4  // Inizializzazione standard della DFS
5  for each vertex u in G.V
6      u.color = BIANCO
7      u.π = NIL
8  
9  // Variabile globale per tenere traccia dei tempi
10 time = 0
11 
12 // Ciclo principale
13 for each vertex u in G.V
14     if u.color == BIANCO
15         DFS-VISIT-TS(G, u, Linked_List)
16

DFS-VISIT-TS(G, u, Linked_List)
1  // u viene appena scoperto
2  time = time + 1
3  u.d = time          // Registriamo il tempo di scoperta
4  u.color = GRIGIO
5  
6  // Esploriamo gli adiacenti
7  for each v in G.Adj[u]
8      if v.color == BIANCO
9          v.π = u
10         DFS-VISIT-TS(G, v, Linked_List)
11         
12 // u ha finito di esplorare tutti i suoi vicini
13 u.color = NERO
14 time = time + 1
15 u.f = time          // Registriamo il tempo di completamento
16 
17 // Inserimento in testa alla lista concatenata
