
MERGESORT(A,p,q)

if (p<q)
    r = [(p+q)/2]
    MERGESORT(A,p,r)
    MERGESORT(A,r+1,q)
    MERGE(A,p,q,r)


MERGE(A,p,q,r)

n1 = q-p+1
n2 = r-q

Siano L[1..n1+1] e R[1..n2+1]

for i = 1 to n1
    L[i] = A[p+i-1]

for j = 1 to n2 
    R[i] = A[q+j]

//Valori sentinella
L[n1+1] = infinito
R[n2+1] = infinito

i = 1
j = 1

for k = p to r 
    if L[i] <= R[i]
        A[k] = L[i]
        i = i+1
    else 
        A[k] = R[j]
        j = j +1
