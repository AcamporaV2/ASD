in pseucodice, che calcoli il numero dei cicli, di lunghezza maggiore di 1, presenti nel grafo



DFS(G)

for each u in V[G]
    padre[u] = NIL 
    color[u] = BIANCO

cicli = 0

for each u in V[G]
    if color == BIANCO
        DFS-VISIT(u ,cicli)

return cicli
