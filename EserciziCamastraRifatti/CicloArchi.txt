Dato un grafo orientato scrivere un algoritmo che restituisca true 
se esiste almeno un ciclo formato da due archi altrimenti restituisca false


DFS(G)
    for each u in V[G]
        color[u] = white 
        pi[u] = NIL 
    time = 0

    for each u in V[G]
        if color[u] == white 
            
            if DFS-VISIT(u) == true 
                return true

    return false 

DFS-VISIT(G, u)
    time = time + 1 
    d[u] = time 
    color[u] = gray 

    for each v in Adj[u]
        // Controllo manuale del ciclo di lunghezza 2 (ignorando i self-loop)
        if u != v
            for each w in Adj[v]
                if w == u
                    return true    // Trovato l'arco di ritorno!

        // Normale prosecuzione della DFS
        if color[v] == white
            pi[v] = u 
            if DFS-VISIT(G, v) == true 
                return true 

    color[u] = black 
    time = time + 1 
    f[u] = time 

    return false



















DFS-VISIT(u)
    time = time +1 
    d[u] = time 
    color[u] = gray 


    for each v in Adj[u]
        if color[u] == white
            pi[v] = u 
            
            if DFS-VISIT(v) == true 
                return true 
        
        else if color[v] == gray 
            if pi[u] == v 
                return true 

    color[u] = black 
    time = time +1 
    f[u] = time 

    return false
