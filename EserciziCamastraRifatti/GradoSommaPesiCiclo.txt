

Dato un grafo orientato ( G = (V, E) ) e pesato con funzione peso ( w : E ->{R} ), 
scrivere in pseudocodice 
un algoritmo ricorsivo che restituisca la somma dei pesi del più lungo ciclo in ( G ).

// Variabile globale per tenere traccia del massimo trovato
max_cycle_weight = -infinity

LONGEST-CYCLE(G, w)
    // Inizializzazione
    for each u in V[G]
        color[u] = white
        d[u] = 0

    // Proviamo a lanciare la ricerca da ogni nodo per essere sicuri
    // di esplorare tutte le componenti e tutte le combinazioni
    for each u in V[G]
        FIND-CYCLE-DFS(u, 0)
        
    // Se è rimasto -infinity, non ci sono cicli
    if max_cycle_weight == -infinity
        return NULL // o 0, a seconda della richiesta
    else
        return max_cycle_weight

FIND-CYCLE-DFS(u, current_dist)
    color[u] = gray
    d[u] = current_dist

    for each v in Adj[u]
        
        // CASO 1: Nodo Bianco -> Continuiamo l'esplorazione
        if color[v] == white
            FIND-CYCLE-DFS(v, current_dist + w(u, v))
        
        // CASO 2: Nodo Grigio -> Ciclo trovato!
        else if color[v] == gray
            // Calcoliamo il peso del ciclo appena chiuso
            // Peso del percorso nello stack (v -> ... -> u) + arco di chiusura (u -> v)
            current_cycle_w = (d[u] - d[v]) + w(u, v)
            
            if current_cycle_w > max_cycle_weight
                max_cycle_weight = current_cycle_w

    // BACKTRACKING
    // Importante: Rimettere a BIANCO permette a questo nodo 
    // di essere attraversato nuovamente da altri percorsi in future iterazioni.
    color[u] = white
