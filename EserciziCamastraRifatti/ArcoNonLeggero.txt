Dato un grafo G = (V, E), scrivere un algoritmo in pseudocodice che restituisca l' arco 
non leggero di peso minimo



KRUSKAL(G,w)

A = insieme vuoto 

for each u in V[G]
    MAKE-SET(u)

Sort(E) 
//ordinati per peso non crescente

for each (u,v) in E[G]

    if FIND-SET(u) != FIND-SET(v)
        A = A u {(u,v)}
        Union(u,v)
    
    else 
        //Il primo arco che forma un ciclo 
        return (u,v)

return NIL
