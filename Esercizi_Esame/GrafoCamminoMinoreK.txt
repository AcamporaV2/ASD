Dato un grafo non orientato $G = (V, E), un nodo sorgente s in V, ed un numero intero k in N, 
scrivere un algoritmo, in pseudocodice, che restituisca il numero dei nodi, il cui cammino minimo dal nodo sorgente s sia minore di k



for each v in V[G] - {s}
    color[v] = white 
    padre[v] = NIL 
    d[v] = Infinito 

color[s] = gray 
padre[s] = 0 
d[s] = 0 

Q = coda vuota 
ENQUEUE(Q,s)


while Q non è vuoto 

    u = DEQUEUE(Q)

    for each v in Adj[u] 
        if color[v] == white 
            color[v] = gray 
            d[v] = d[u] + 1
            padre[v] = u 
            ENQUEUE(Q,v)
    
    color[u] = black 

count = 0 

for each u in V[G]
    if d[u] < k 
        count = count +1 

return count
