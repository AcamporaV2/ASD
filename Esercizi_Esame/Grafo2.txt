// Funzione Principale
HAMILTONIAN-CYCLE(G)
  n = G.V.length
  let path[1..n] be a new array
  let visited[1..n] be a new boolean array
  
  for i = 1 to n
      visited[i] = FALSE

  visited[1] = TRUE

// Funzione Ricorsiva di Backtracking
// k è la posizione corrente nel percorso da riempire
HAMILTONIAN-CYCLE-VISIT(G, path, visited, k)
  n = G.V.length
