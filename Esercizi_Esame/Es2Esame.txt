Edge_MST <- KRUSKAL(G,w)
Edge_NonSicuri <- VUOTO

for each edge e ∈ E[G]
    if e ∉ Edge_MST
        Edge_NonSicuri <- Edge_NonSicuri U {e}

if Edge_NonSicuri = VUOTO
    return NIL

//Ordina gli archi in ordine non decrescente di peso W
n <- |Edge_NonSicuri|
mediana <- (n/2)

Edge_mediano <- Edge_NonSicuri[mediana]


return Edge_mediano
