RELAX(u, v, w)
1  if d[v] > d[u] + w(u, v)                     // Se passare per u migliora il cammino
2      d[v] ← d[u] + w(u, v)                    // Aggiorna distanza di v
3      π[v] ← u                                 // u diventa predecessore di v
4      DECREASE-KEY(Q, v, d[v])                 // Aggiorna priorità di v in coda


EXTRACT-MIN(Q)
1  min ← Q[1]                                   // Salva l'elemento minimo (radice heap)
2  Q[1] ← Q[heap-size[Q]]                       // Sposta ultimo elemento in radice
