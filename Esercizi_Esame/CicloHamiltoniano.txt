// Funzione di utilità ricorsiva
HAMILTONIAN-CYCLE-VISIT(G, u, path, visited, k)
1  n = G.V.length
2  // Se tutti i vertici sono nel percorso
3  if k == n
4      // Controlla se c'è un arco dall'ultimo vertice al primo
5      if (path[n], path[1]) is in G.E
6          return TRUE // Ciclo trovato
7      else
8          return FALSE
9
10 // Prova tutti i vertici adiacenti a u
11 for each v in G.Adj[u]
12     if visited[v] == FALSE
13         visited[v] = TRUE
14         path[k+1] = v
15         // Chiamata ricorsiva
16         if HAMILTONIAN-CYCLE-VISIT(G, v, path, visited, k + 1)
17             return TRUE
18         // Backtrack
19         visited[v] = FALSE
20
21 return FALSE

// Funzione principale
FIND-HAMILTONIAN-CYCLE(G)
1  n = G.V.length
2  let path be a new array of size n + 1
3  let visited be a new array of size n
4  for i = 1 to n
5      visited[i] = FALSE
6
7  // Inizia dal primo vertice (la scelta è arbitraria)
8  start_v = G.V[1]
9  path[1] = start_v
10 visited[start_v] = TRUE
11
12 // Avvia la ricerca ricorsiva
13 if HAMILTONIAN-CYCLE-VISIT(G, start_v, path, visited, 1)
14     return TRUE
15 else
16     return FALSE
