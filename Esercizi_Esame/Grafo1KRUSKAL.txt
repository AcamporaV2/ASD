/*
Dato un grafo G = (V, E), scrivere un algoritmo in pseudocodice che restituisca l' arco 
non leggero di peso minimo
*/



TROVA-MIN (G, w)
T <- KRUSKAL-MST(G,w)
min_peso <- infinito
min_edge <- NULL

per ogni edge(u,v) in E[G] Do 
    if(u,v) non appartiene T allora
        if w(u,v) < min_peso allora
            min_peso <- w(u,v)
            min_edge <- (u,v)

return min_edge
