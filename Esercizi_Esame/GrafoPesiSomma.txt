/*


Dato un grafo orientato ( G = (V, E) ) e pesato con funzione peso ( w : E ->{R} ), 
scrivere in pseudocodice 
un algoritmo ricorsivo che restituisca la somma dei pesi del più lungo ciclo in ( G ).



MaxPeso <- -infinito

TROVA-CICLO-MAX (G, w)
  for each s in V[G]
    Visitate <- insieme vuoto
    DFS-CICLO(s, s, 0, Visitate)
  return MaxPeso


DFS-CICLO (start, u, pesoCorr, Visitate)
  Visitate = Visitate ∪ {u}  // Marco u come parte del cammino corrente

  for each v in Adj[u]
    
    // Caso Base: se il vicino 'v' è il nodo di partenza 'start', ho chiuso un ciclo
    if v == start
      if pesoCorr + w(u,v) > MaxPeso
         MaxPeso = pesoCorr + w(u,v)

    // Passo Ricorsivo: se 'v' non è già nel cammino corrente, continuo a esplorare
    else if v ∉ Visitate
      DFS-CICLO(start, v, pesoCorr + w(u,v), Visitate)

  // Backtracking: rimuovo u dai visitati prima di risalire
  // Questo permette di trovare cicli diversi che passano per lo stesso nodo
  Visitate = Visitate - {u}
