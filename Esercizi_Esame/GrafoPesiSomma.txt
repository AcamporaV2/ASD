/*


Dato un grafo orientato ( G = (V, E) ) e pesato con funzione peso ( w : E ->{R} ), 
scrivere in pseudocodice 
un algoritmo ricorsivo che restituisca la somma dei pesi del più lungo ciclo in ( G ).



DFS-LUNGO-CICLO(G)
    Peso_max ← -∞
    for each s in V[G]
        for each u in V[G]
            color[u] ← WHITE
        DFS-VISITA(G, s, s, 0, Peso_max)
    return Peso_max


DFS-VISITA(G, s, u, peso_corrente, Peso_max)
    color[u] ← GRAY

    for each v in Adj[u]
        if v == s then
            if peso_corrente + w(u,v) > Peso_max then
                Peso_max ← peso_corrente + w(u,v)
        else if color[v] == WHITE then
            DFS-VISITA(G, s, v, peso_corrente + w(u,v), Peso_max)

    color[u] ← WHITE   // backtrack
