Dato una coda di min priorità, implementata mediante un heap, scrivere un algoritmo che restituisca, 
come output, il numero degli scambi necessari ad eseguire il metodo EXTRACT_MIN.

HEAP

HEAP-EXTRACT-MIN-COUNT(A)
    // Controllo Underflow
    if A.heap-size < 1
        error "heap underflow"
    
    // Salviamo il minimo (non serve al conteggio, ma è standard)
    min = A[1]
    
    // Spostiamo l'ultimo elemento nella radice
    A[1] = A[A.heap-size]
    
    // Riduciamo la dimensione dello heap
    A.heap-size = A.heap-size - 1
    
    // Chiamiamo la Min-Heapify modificata per contare gli scambi
    // e ritorniamo il risultato
    num_scambi = MIN-HEAPIFY-COUNT(A, 1)
    
    return num_scambi


MIN-HEAPIFY-COUNT(A, i)
    l = LEFT(i)    // Indice del figlio sinistro (2*i)
    r = RIGHT(i)   // Indice del figlio destro (2*i + 1)
    smallest = i
    
    // Determina il più piccolo tra i, sinistro e destro
    if l <= A.heap-size and A[l] < A[i]
        smallest = l
    
    if r <= A.heap-size and A[r] < A[smallest]
        smallest = r
        
    // Se l'elemento i non è il più piccolo, dobbiamo scambiare
    if smallest != i
        exchange A[i] with A[smallest]
        
        // Restituiamo 1 (lo scambio corrente) + gli scambi fatti ricorsivamente
        return 1 + MIN-HEAPIFY-COUNT(A, smallest)
        
    else
        // Caso base: la proprietà di min-heap è rispettata, 0 scambi
        return 0
