/*

Dato un grafo non orientato G=(V,E) si scriva un algoritmo che calcoli il MST 
e ne restituisca l'arco all'interno del MST di peso mediano. 
Si ricordi che il mediano di un insieme è l'elemento in cui valore è maggiore e minore della parte intera degli elementi. 
Per semplicità si assuma che la cardinalità dell'insieme dei vertici di V sia pari.




KRUSKAL(G)

    // FASE 1: Calcola il MST usando Kruskal
    A = ∅
    
    for each v in V[G]
        MAKE-SET(v)
    
    // Ordina gli archi per peso non decrescente
    Ordina E[G] per w(u,v) non decrescente
    
    for each (u,v) in E[G] (in ordine di peso)
        if FIND-SET(u) ≠ FIND-SET(v)
            A = A ∪ {(u,v)}
            UNION(u,v)
    
    // FASE 2: Trova l'arco mediano nel MST
    // A contiene |V| - 1 archi
    
    // Converti A in array e ordina per peso
    L = array contenente gli archi di A
    Ordina L per w(u,v) non decrescente
    
    // Calcola l'indice mediano
    n = |L|
    indice_mediano = ⌊n / 2⌋
    
    // Restituisci l'arco mediano
    return L[indice_mediano]
