PRIM-HEIGHT(G, w, r)
    // Inizializzazione
    for each v in V[G] - {r}
        key[v] = ∞
        π[v] = NIL
        H[v] = 0

    key[r] = 0
    π[r] = NIL
    H[r] = 0

    Q = V[G]
    max_alt = 0

    // Ciclo principale
    while Q ≠ ∅
        u = EXTRACT-MIN(Q)



        // Scansione adiacenti e relax
        for each v in Adj[u]
            if v ∈ Q and w(u, v) < key[v]
                key[v] = w(u, v)
                π[v] = u
                // (Implica un DECREASE-KEY su Q)
        // Calcolo dell'altezza (π[u] è già definitivo e fuori da Q)
        if π[u] ≠ NIL
            H[u] = H[π[u]] + 1

        // Aggiornamento altezza massima
        if H[u] > max_alt
            max_alt = H[u]
    return max_alt
