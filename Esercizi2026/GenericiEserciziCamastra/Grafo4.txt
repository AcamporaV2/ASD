vere un algoritmo, in pseudocodice, che restituisca la lista dei nodi, il cui
cammino minimo dal nodo sorgente s é massimo. (7 punti)


for each v in V[G]
    color[v] = white 
    padre[v] = NIL 
    d[v] = Infinito

color[s] = grey 
d[s] = 0 

max = - 1
Nodi= {}
Q = coda vuota 
ENQUEUE(Q,s)

while Q non è vuota 

    u = DEQUEUE(Q)

    for each v in Adj[u]
        if color[v] == white 
            color[v] = grey 
            d[v] = d[u] + 1
            padre[v] = u 
            ENQUEUE(Q,v)
        
        if d[v] > max 
            max = d[v] 
            Nodi = {v}
        else if d[v] == max 
            Nodi = Nodu U {v}

    color[u] = black
