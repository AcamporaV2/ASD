Dato un grafo G=(V,E), una sorgente s e un peso W Lista dei nodi foglia dell'albero dei cammini minimi


Dijkstra(G,w,s)

for each v in V[G] - {s}
    d[v] = infinito
    padre[v] = NIL 
    
d[s] = 0
padre[s] = NIL 

S = {}
Q = V[G]

while Q non è vuota 

    u = EXTRACT_MIN(Q)

    for each v in Adj[u]
        if d[v] > d[u] + w(u,v)
            d[v] = d[u] + w(u,v)
            padre[v] = u 



sia Parente[1...|V[G]|] un nuovo array booleano

for each v in V[G]
    Parente[v] = FALSE 

for each v in V[G]
    if padre[v] != nil 
        Parente[padre[v]] = TRUE 

Foglie = nuova lista vuota 
for each v in V[G]
    if d[v] < infinito and Parente[v] == FALSE
        Inserisci v in Foglie 

Return foglie
