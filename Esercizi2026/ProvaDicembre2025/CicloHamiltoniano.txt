HAS-HAMILTONIAN-CYCLE(G)
1  // Un ciclo in un grafo semplice richiede almeno 3 vertici
2  se |G.V| < 3
3      ritorna FALSE
4      
5  // Inizializzazione: nessun vertice è nel cammino
6  per ogni vertice u ∈ G.V
7      u.color = WHITE
8      
9  // Scegliamo un vertice di partenza arbitrario
10 s = primo vertice di G.V 
11 s.color = GRAY
12 
13 // Avviamo la ricerca ricorsiva passando: grafo, nodo corrente, nodo di partenza, contatore nodi
14 ritorna HAMILTONIAN-VISIT(G, s, s, 1)


HAMILTONIAN-VISIT(G, u, start, count)
1  // Caso base: abbiamo visitato tutti i vertici del grafo
2  se count == |G.V|
3      // Controlliamo se c'è un arco che chiude il ciclo verso il nodo di partenza
4      se start ∈ G.Adj[u]
5          ritorna TRUE
6      altrimenti
7          ritorna FALSE
8          
9  // Passo ricorsivo: esploriamo i vicini del nodo corrente
10 per ogni vertice v ∈ G.Adj[u]
11     se v.color == WHITE
12         v.color = GRAY       // Aggiungiamo 'v' al cammino corrente
13         
14         // Se l'esplorazione da 'v' trova un ciclo, propaghiamo il successo
15         se HAMILTONIAN-VISIT(G, v, start, count + 1) == TRUE
16             ritorna TRUE
17             
18         // Backtracking: se 'v' non porta a una soluzione, lo togliamo dal cammino
19         v.color = WHITE      
20 
21 // Se nessun vicino porta a un ciclo Hamiltoniano
22 ritorna FALSE
